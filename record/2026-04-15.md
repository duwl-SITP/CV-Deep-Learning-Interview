## 1. 手撕Multi-head Attention

``` python
class MultiHeadAttn(nn.Module):
	def __init__(self, dim, num_heads):
		self.q = nn.Linear(dim, dim)
		self.k = nn.Linear(dim, dim)
		self.v = nn.Linear(dim, dim)
		self.out = nn.Linear(dim, dim)
		
		self.num_heads = num_heads
		self.d_model = dim // num_heads
		
	def forward(self, x, mask=None):
		B, N, C = x.shape
		
		q = self.q(x)
		k = self.k(x)
		v = self.v(x)
		
		q = q.view(B, N, self.num_heads, self.d_model).transpose(1, 2)
		k = k.view(B, N, self.num_heads, self.d_model).transpose(1, 2)
		v = v.view(B, N, self.num_heads, self.d_model).transpose(1, 2)
		
		attn = torch.matmul(q, k.transpose(-1, -2)) / math.sqrt(self.d_model)
		
		if mask is not None:
			attn = attn.masked_fill(mask==0, float('-inf'))
		attn = torch.softmax(attn, dim=-1)
		
		scores = torch.matmul(attn, v)
		scores = scores.transpose(1, 2).contiguous().view(B, N, C)
		out = self.out(scores)
		return out
```

## 1. 手撕RoPE旋转位置编码

``` python
class RoPE(nn.Module):
	def __init__(self, dim, max_pos = 2048):
		super().__init__()
		self.dim = dim		
		self.max_pos = max_pos
		
		# pos
		pos = torch.arange(0, max_pos) 
		# c / 2
		freq = 1 / 10000 ** (torch.arange(0, dim, 2).float() / dim)
		# pos * dim/2
		self.theta = pos[:, None] * freq[None, :]
		# register_buffer使其作为不可训练参数
		self.register_buffer("cos_freq", torch.cos(self.theta))  
		self.register_buffer("sin_freq", torch.sin(self.theta))
		
	def forward(self, x):
		B, L, D = x.shape  
		x1 = x[..., 0::2]  
		x2 = x[..., 1::2]		
		
		# 长度维度匹配
		cos = self.cos_freq[:L].unsqueeze(0) # [1, L, D/2]  
		sin = self.sin_freq[:L].unsqueeze(0)
		
		x1_rot = x1 * cos - x2 * sin
		x2_rot = x1 * sin + x2 * cos
		
		x_rot = torch.stack([x1_rot, x2_rot], dim=-1)  
		x_rot = x_rot.flatten(-2)
		return x_rot
```

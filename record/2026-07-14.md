## 手撕KL散度

``` python
import torch

def kl_divergence_prob(
    p: torch.Tensor,
    q: torch.Tensor,
    dim: int = -1,
    reduction: str = "batchmean",
    eps: float = 1e-8,
) -> torch.Tensor:
    """
    计算 KL(P || Q)。

    Args:
        p: 概率分布 P，shape 通常为 [batch_size, num_classes]
        q: 概率分布 Q，shape 与 p 相同
        dim: 类别所在维度
        reduction:
            "none"      返回每个样本的 KL
            "sum"       所有样本 KL 求和
            "mean"      所有样本 KL 求平均
            "batchmean" 所有 KL 元素求和后除以 batch_size
        eps: 防止 log(0)

    Returns:
        KL(P || Q)
    """
    p = p.clamp_min(eps)
    q = q.clamp_min(eps)

    # 对类别维度求和，得到每个样本的 KL
    kl_per_sample = torch.sum(
        p * (torch.log(p) - torch.log(q)),
        dim=dim,
    )

    if reduction == "none":
        return kl_per_sample
    elif reduction == "sum":
        return kl_per_sample.sum()
    elif reduction == "mean":
        return kl_per_sample.mean()
    elif reduction == "batchmean":
        # 当每个样本只有一个类别维度时，与 mean 等价
        return kl_per_sample.sum() / p.shape[0]
    else:
        raise ValueError(f"Unsupported reduction: {reduction}")
```

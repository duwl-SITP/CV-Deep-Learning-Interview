## 1. 手撕交叉熵损失函数

``` python
def binary_cross_entropy(pred: Tensor, label: Tensor):
	epsilon = 1e-12  
	pred = pred.clamp(pred, min=epsilon, max=1.)
	loss = torch.mean(-label*torch.log(pred) - (1-label)*torch.log((1-pred)))
	return loss
	
def cross_entropy(pred: Tensor, label: Tensor):
	epsilon = 1e-12
	# 1️⃣ softmax 先做  
	probs = torch.softmax(pred, dim=-1)  
	  
	# 2️⃣ clamp 防止 log(0)  
	probs = torch.clamp(probs, min=1e-12, max=1.0)
	
	one_hot = torch.zeros_like(probs)
	idx = torch.arange(len(label), device=label.device)
	one_hot[idx, label] = 1
	
	loss = torch.mean(torch.sum(-one_hot * torch.log(probs), dim = -1))
	return loss
```

## 2. 手撕Focal loss损失函数

``` python
def focal_loss(pred: Tensor, label: Tensor, alpha=0.25, gamma=2.0):
	epsilon = 1e-12
	# 1️⃣ softmax 先做  
	probs = torch.softmax(pred, dim=-1)  
	  
	# 2️⃣ clamp 防止 log(0)  
	probs = torch.clamp(probs, min=1e-12, max=1.0)
	
	idx = torch.arange(len(label), device=label.device)
	pt = probs[idx, label]  
  
	# alpha_t  
	if isinstance(alpha, torch.Tensor):  
		alpha_t = alpha[label]  
	else:  
		alpha_t = alpha  
  
	loss = -alpha_t * (1 - pt) ** gamma * torch.log(pt)  
	return loss.mean()
```

## 3. 手撕BatchNorm

``` python
class BatchNorm(nn.Module):
	def __init__(self, inc, eps=1e-5, momentum=0.1):
		super().__init__()
		self.eps = eps  
		self.momentum = momentum

		self.weight = nn.Parameter(torch.ones(inc))  
		self.bias = nn.Parameter(torch.zeros(inc))
		
		self.register_buffer("running_mean", torch.zeros(inc))  
		self.register_buffer("running_var", torch.ones(inc))
	
	def forward(self, x):
		if self.training:
			mu = x.mean(dim=(0, 2, 3))
			var = x.var(dim=(0, 2, 3))
			self.running_mean = self.momentum * mu + (1-self.momentum) * self.running_mean
			self.running_var = self.momentum * var + (1-self.momentum) * self.running_var
		else:
			mu = self.running_mean
			var = self.running_var
		
		x_hat = (x - mu[None, :, None, None]) / torch.sqrt(var[None, :, None, None] + self.eps)
		x_hat = x_hat * self.weight[None, :, None, None] + self.bias[None, :, None, None]
		return x_hat
```

## 3. 手撕LayerNorm

``` python
class LayerNorm(nn.Module):
	def __init__(self, inc, eps):
		super().__init__()
		self.eps = eps
		self.weight = nn.Parameter(torch.ones(inc))  
		self.bias = nn.Parameter(torch.zeros(inc))
		
	def forward(self, x):
		mu = x.mean(dim=-1, keepdim=True)  
		var = x.var(dim=-1, keepdim=True)  
		  
		x_hat = (x - mu) / torch.sqrt(var + self.eps)  
		x_hat = x_hat * self.weight + self.bias
		return x_hat
```
